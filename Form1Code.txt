namespace WinFormsApp1
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void btnPlay_Click(object sender, EventArgs e)
        {
            int treasure = 1;

            lstScore.Items.Clear();

            while (treasure <= 10)
            {
                lstScore.Items.Add("Treasure " + treasure + " Collected! ");
                treasure++;
            }
        }
    }
}
