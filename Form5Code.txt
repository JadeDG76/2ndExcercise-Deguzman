using System;
using System.Collections.Generic;
using System.ComponentModel;
using System.Data;
using System.Drawing;
using System.Text;
using System.Windows.Forms;

namespace WinFormsApp1
{
    public partial class Form5 : Form
    {
        public Form5()
        {
            InitializeComponent();
        }

        private void btnFight_Click(object sender, EventArgs e)
        {
            string weapon = cmbWeapon.Text;

            switch (weapon)
            {
                case "Sword":
                    lblResult.Text = "⚔️ You attacked with a Sword!";
                    break;

                case "Bow":
                    lblResult.Text = "🏹 You attacked with a Bow!";
                    break;

                case "Magic":
                    lblResult.Text = "✨ You cast a Magic Spell!";
                    break;

                default:
                    lblResult.Text = "Select a weapon first!";
                    break;
            }
            }
    }
}
